31Matrix<T> numpy_to_matrix(py::array_t<T, py::array::c_style | py::array::forcecast> array) {
32 py::buffer_info buf = array.request();
33
34 if (buf.ndim != 2) {
35 throw std::runtime_error("Input must be a 2D array");
36 }
37
38 size_t rows = buf.shape[0];
39 size_t cols = buf.shape[1];
40 T* ptr = static_cast<T*>(buf.ptr);

57template<typename T>
58py::array_t<T> matrix_to_numpy_copy(const Matrix<T>& matrix) {
59 // Create output array with proper shape
60 py::array_t<T> result({static_cast<py::ssize_t>(matrix.rows()),
61 static_cast<py::ssize_t>(matrix.cols())});
62
63 py::buffer_info buf = result.request();
64 T* ptr = static_cast<T*>(buf.ptr);
65
66 // Fast memory copy from Matrix to NumPy buffer
67 std::memcpy(ptr, matrix.data(), sizeof(T) * matrix.rows() * matrix.cols());
68
69 return result;

86template<typename T>
87py::array_t<T> matrix_to_numpy_zerocopy(Matrix<T>&& matrix) {
88 // Move matrix to heap so it outlives this function
89 auto* matrix_ptr = new Matrix<T>(std::move(matrix));
90
91 // Create capsule that will delete the Matrix when NumPy array is destroyed
92 py::capsule free_when_done(matrix_ptr, [](void* p) {
93 delete reinterpret_cast<Matrix<T>*>(p);
94 });
95
96 return py::array_t<T>(
97 {static_cast<py::ssize_t>(matrix_ptr->rows()),
98 static_cast<py::ssize_t>(matrix_ptr->cols())}, // shape
99 {static_cast<py::ssize_t>(matrix_ptr->cols() * sizeof(T)),
100 static_cast<py::ssize_t>(sizeof(T))}, // strides (row-major)
101 matrix_ptr->data(), // data pointer
