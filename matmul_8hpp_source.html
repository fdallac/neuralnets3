74 static Matrix<T> mm(const Matrix<T>& A, const Matrix<T>& B,
75 MatMulMethod method = MatMulMethod::Auto) {
76
77 if (method == MatMulMethod::Auto) {
78 // Heuristic for automatic method selection
79 std::size_t volume = A.rows() * A.cols() * B.cols();
80
81 if (volume <= 128 * 128 * 128) {
82 // Small matrices: use SIMD implementation
83 method = MatMulMethod::SIMD_AVX512;
84 } else if (volume <= 1024 * 1024 * 1024) {
85 // Medium matrices: use pure CUDA if available, else SIMD + OpenMP
86#ifdef CUDA_AVAILABLE
87 method = MatMulMethod::CUDA;
88#else
89 method = MatMulMethod::SIMD_OpenMP_Tile;
90#endif
91 }
92 } else {
93 // Large matrices: use hybrid CUDA + OpenMP if available, else SIMD + OpenMP
94#ifdef CUDA_AVAILABLE
95 method = MatMulMethod::CUDA_OpenMP;
96#else
97 method = MatMulMethod::SIMD_OpenMP_Tile;
98#endif
99 }
100
101 // Dispatch to selected method
102 switch (method) {
103 case MatMulMethod::Vanilla:
104 return mm_vanilla(A, B);
105
106 case MatMulMethod::LoopUnrolled4:
107 return mm_unrolled4(A, B);
108
109 case MatMulMethod::LoopUnrolled8:
110 return mm_unrolled8(A, B);
111
112 case MatMulMethod::LoopUnrolled:
113 return mm_unrolled<8>(A, B);
114
115 case MatMulMethod::Tiled:
116 return mm_tiled<16>(A, B);
117
118 case MatMulMethod::OpenMP:
119 return mm_openmp(A, B);
120
121 case MatMulMethod::SIMD_AVX512:
122 return mm_avx512(A, B);
123
124 case MatMulMethod::SIMD_OpenMP_Tile:
125 return mm_simd_openmp_tile(A, B);
126
127 case MatMulMethod::CUDA:
128#ifdef CUDA_AVAILABLE
129 return mm_cuda(A, B);
130#else
131 throw std::invalid_argument("CUDA method selected but not compiled with nvcc");
132#endif
133
134 case MatMulMethod::CUDA_OpenMP:
135#ifdef CUDA_AVAILABLE
136 return mm_cuda_openmp(A, B);
137#else
138 throw std::invalid_argument("CUDA_OpenMP method selected but not compiled with nvcc");
139#endif
140
141 // Future methods can be added here
142 default:
143 throw std::invalid_argument("Unknown matrix multiplication method");
144 }
145 }

205 static Matrix<T> mm_unrolled4(const Matrix<T>& A, const Matrix<T>& B) {
206 if (A.cols() != B.rows()) {

208 }

210 const std::size_t N = A.cols();
211 for (std::size_t i = 0; i < A.rows(); ++i) {
212 for (std::size_t j = 0; j < B.cols(); ++j) {
213 T acc0 = T{};
214 T acc1 = T{};
215 T acc2 = T{};
216 T acc3 = T{};
217 std::size_t k = 0;
218 // Process 4 elements at a time
219 for (; k + 3 < N; k += 4) {
220 acc0 += A(i, k ) * B(k, j);
221 acc1 += A(i, k + 1) * B(k + 1, j);
222 acc2 += A(i, k + 2) * B(k + 2, j);
223 acc3 += A(i, k + 3) * B(k + 3, j);
224 }
225 // Add the partial sums together

227 // Handle leftover values
228 for (; k < N; ++k) {

230 }
231 C(i, j) = sum;

511 static void mm_avx512_transpose_int(const int* A, const int* B_t, int* C, std::size_t M, std::size_t N, std::size_t K) {
512 constexpr std::size_t V = 16;
513 for (std::size_t i = 0; i < M; ++i) {
514 const int* arow = A + i * K;
515 int* crow = C + i * N;
516 for (std::size_t j = 0; j < N; ++j) {
517 const int* brow = B_t + j * K;
518 __m512i acc = _mm512_setzero_si512();
519 std::size_t k = 0;
520 for (; k + V <= K; k += V) {
521 __m512i va = _mm512_loadu_si512((__m512i const*)(arow + k));
522 __m512i vb = _mm512_loadu_si512((__m512i const*)(brow + k));
523 acc = _mm512_add_epi32(acc, _mm512_mullo_epi32(va, vb));
524 }
525 int tmp[V];
526 _mm512_storeu_si512((__m512i*)tmp, acc);
527 int sum = 0;
528 for (std::size_t t = 0; t < V; ++t) sum += tmp[t];
529 for (; k < K; ++k) sum += arow[k] * brow[k];
530 crow[j] = sum;
531 }
532 }

683 const std::size_t M = A.rows();
684 const std::size_t N = B.cols();
685 const std::size_t K = A.cols();
686 const T* A_data = A.data();
687 const T* B_data = B.data();
688 T* C_data = C.data();
689
690
691#if !defined(_OPENMP)
692 // Warning if openmp is not enabled
693 #warning "OpenMP is not enabled. Compiling without OpenMP support."
694#endif
695
696 // Set number of threads
697#if defined(_OPENMP)
698 if (num_threads <= 0) {
699 num_threads = omp_get_max_threads();
700 }
701 omp_set_num_threads(num_threads);
702
703 // Parallelize outer loop with OpenMP
704 #pragma omp parallel for schedule(static)
705#endif
706 for (std::size_t i = 0; i < M; ++i) {
707 for (std::size_t j = 0; j < N; ++j) {
708 T sum = T{};
709 for (std::size_t k = 0; k < K; k += block_k) {
710 std::size_t k_max = std::min(k + block_k, K);
711 for (std::size_t kk = k; kk < k_max; ++kk) {
712 sum += A_data[i * K + kk] * B_data[kk * N + j];
713 }
714 }
715 C_data[i * N + j] = sum;
716 }
