268class Softmax : public Activation<T> {
269 public:
276 Matrix<T> forward(const Matrix<T>& Z) {
277 Matrix<T> result(Z.rows(), Z.cols());
278 for (std::size_t i = 0; i < Z.rows(); ++i) {
279 T max_val = Z(i, 0);
280 for (std::size_t j = 1; j < Z.cols(); ++j) {
281 if (Z(i, j) > max_val) {
282 max_val = Z(i, j);
283 }
284 }
285 T sum_exp = T{};
286 for (std::size_t j = 0; j < Z.cols(); ++j) {
287 result(i, j) = std::exp(Z(i, j) - max_val);
288 sum_exp += result(i, j);
289 }
290 for (std::size_t j = 0; j < Z.cols(); ++j) {
291 result(i, j) /= sum_exp;
292 }
293 }
294 return result;
295 }

305 Matrix<T> backward(const Matrix<T>& Z, const Matrix<T>& d_A) {
306 // Recompute Softmax(Z) -> S
307 Matrix<T> S = forward(Z);
308 Matrix<T> d_Z(Z.rows(), Z.cols());
309
310 // Compute Gradient: dZ_i = S_i * (dA_i - Sum(dA_k * S_k))
311 for (size_t i = 0; i < Z.rows(); ++i) {
312
313 // Calculate dot product (dA . S) for this row
314 T dot = static_cast<T>(0);
315 for (size_t k = 0; k < Z.cols(); ++k) {
316 dot += d_A(i, k) * S(i, k);
317 }
318
319 // Apply formula
320 for (size_t j = 0; j < Z.cols(); ++j) {
321 d_Z(i, j) = S(i, j) * (d_A(i, j) - dot);
322 }
323 }
324 return d_Z;
325 }
