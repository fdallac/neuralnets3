155class Adam : public Optimizer<T> {
156 public:
164 Adam(T learning_rate, T beta1 = 0.9, T beta2 = 0.999, T epsilon = 1e-8)
165 : learning_rate(learning_rate), beta1(beta1), beta2(beta2), epsilon(epsilon), ts(1) {}
166
171 T get_learning_rate() const { return learning_rate; }

181 void update(Matrix<T>& parameter, const Matrix<T>& gradient, Matrix<T>& momentum, Matrix<T>& velocity, const std::string& layer_id = "") {
182 for (std::size_t i = 0; i < parameter.rows(); ++i) {
183 for (std::size_t j = 0; j < parameter.cols(); ++j) {
184 // Update biased first moment estimate
185 momentum(i, j) = beta1 * momentum(i, j) + (1 - beta1) * gradient(i, j);
186 // Update biased second moment estimate
187 velocity(i, j) = beta2 * velocity(i, j) + (1 - beta2) * gradient(i, j) * gradient(i, j);
188
189 // Compute bias-corrected first and second moment estimates
190 T m_hat = momentum(i, j) / (1 - std::pow(beta1, ts));
191 T v_hat = velocity(i, j) / (1 - std::pow(beta2, ts));
192
193 // Update parameters
194 parameter(i, j) -= learning_rate * m_hat / (std::sqrt(v_hat) + epsilon);
195 }
196 }
197 }
