90 }
91 }

202 }

217 }
218 return static_cast<double>(true_positive) / static_cast<double>(true_positive + false_positive);

287template<typename T>
288inline std::size_t argmax_row(const Matrix<T>& mat, std::size_t row) {
289 std::size_t max_idx = 0;
290 T max_val = mat(row, 0);
291 for (std::size_t j = 1; j < mat.cols(); ++j) {
292 if (mat(row, j) > max_val) {
293 max_val = mat(row, j);
294 max_idx = j;
295 }
296 }
297 return max_idx;
298}

374template<typename T>
375class MultiClassPrecision : public Metrics<T> {
376 public:
383 double eval(const Matrix<T>& y_pred, const Matrix<T>& y_true) override {
384 if (y_pred.rows() != y_true.rows()) {
385 throw std::invalid_argument("Number of samples must match for multi-class precision");
386 }
387
388 // Find number of classes
389 int max_class = 0;
390 for (std::size_t i = 0; i < y_true.rows(); ++i) {
391 max_class = std::max(max_class, static_cast<int>(y_true(i, 0)));
392 max_class = std::max(max_class, static_cast<int>(y_pred(i, 0)));
393 }
394 std::size_t num_classes = static_cast<std::size_t>(max_class + 1);
395
396 // Count TP and FP for each class
397 std::vector<std::size_t> true_positives(num_classes, 0);
398 std::vector<std::size_t> false_positives(num_classes, 0);
399
400 for (std::size_t i = 0; i < y_true.rows(); ++i) {
401 std::size_t pred_class = static_cast<std::size_t>(y_pred(i, 0));
402 std::size_t true_class = static_cast<std::size_t>(y_true(i, 0));
403
404 if (pred_class == true_class) {
405 true_positives[pred_class]++;
406 } else {
407 false_positives[pred_class]++;
408 }
409 }
410
411 // Compute macro-averaged precision
412 double total_precision = 0.0;
413 std::size_t valid_classes = 0;
414
415 for (std::size_t c = 0; c < num_classes; ++c) {
416 std::size_t total_predicted = true_positives[c] + false_positives[c];
417 if (total_predicted > 0) {
418 total_precision += static_cast<double>(true_positives[c]) / static_cast<double>(total_predicted);
419 ++valid_classes;
420 }
421 }
422
423 return valid_classes > 0 ? total_precision / static_cast<double>(valid_classes) : 0.0;
424 }
