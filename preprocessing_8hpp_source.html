55 OneHotEncoder& fit(const Matrix<T>& y) {
56 if (y.cols() != 1) {
57 throw std::invalid_argument("OneHotEncoder expects a column vector (n_samples x 1)");
58 }
59
60 // Collect unique classes
61 std::vector<T> unique_values;
62 for (std::size_t i = 0; i < y.rows(); ++i) {
63 T val = y(i, 0);
64 if (std::find(unique_values.begin(), unique_values.end(), val) == unique_values.end()) {
65 unique_values.push_back(val);
66 }
67 }
68
69 // Sort classes for consistent ordering
70 std::sort(unique_values.begin(), unique_values.end());
71 classes_ = unique_values;
72 num_classes_ = classes_.size();

74 // Build class-to-index mapping
75 class_to_index_.clear();
76 for (std::size_t i = 0; i < classes_.size(); ++i) {
77 class_to_index_[classes_[i]] = i;
78 }
79
80 fitted_ = true;
81 return *this;

93 Matrix<T> transform(const Matrix<T>& y) const {
94 if (!fitted_) {
95 throw std::runtime_error("OneHotEncoder has not been fitted. Call fit() first.");
96 }
97 if (y.cols() != 1) {
98 throw std::invalid_argument("OneHotEncoder expects a column vector (n_samples x 1)");
99 }
100
101 std::size_t n_samples = y.rows();
102 Matrix<T> encoded(n_samples, num_classes_);
103
104 // Initialize to zeros
105 for (std::size_t i = 0; i < n_samples; ++i) {
106 for (std::size_t j = 0; j < num_classes_; ++j) {
107 encoded(i, j) = T{0};
108 }
109 }
110
111 // Set one-hot values
112 for (std::size_t i = 0; i < n_samples; ++i) {
113 T val = y(i, 0);
114 auto it = class_to_index_.find(val);
115 if (it == class_to_index_.end()) {
116 throw std::out_of_range("Unknown class value encountered during transform: " + std::to_string(static_cast<int>(val)));
117 }
118 encoded(i, it->second) = T{1};
119 }
