51 NeuralLayer(std::size_t input_size, std::size_t output_size, Activation<T>& activation, Normalization<T>* normalization = nullptr, std::string id = "")
52 : activation(activation), normalization(normalization), _id(id) {
53 // Weights and bias are initialized to zero by default
54 this->W = Matrix<T>(input_size, output_size);

61 // Initialize weights with He initialization (good for ReLU)

69 this->_id = std::to_string(rand());

82 void normally_initialize_weights(T mean, T stddev) {
83 for (std::size_t i = 0; i < W.rows(); ++i) {
84 for (std::size_t j = 0; j < W.cols(); ++j) {
85 // Random value in [-1, 1] range, then scale by stddev
86 T random_val = static_cast<T>(2.0) * static_cast<T>(rand()) / static_cast<T>(RAND_MAX) - static_cast<T>(1.0);
87 W(i, j) = mean + stddev * random_val;
88 }
89 }

99 void normally_initialize_bias(T mean, T stddev) {
100 for (std::size_t j = 0; j < b.cols(); ++j) {
101 // Random value in [-1, 1] range, then scale by stddev
102 T random_val = static_cast<T>(2.0) * static_cast<T>(rand()) / static_cast<T>(RAND_MAX) - static_cast<T>(1.0);
103 b(0, j) = mean + stddev * random_val;

168 void train(const Matrix<T>& X, const Matrix<T>& y, int epochs, bool verbose = true) {
169
170 // Simple training loop
171 for (int epoch = 0; epoch < epochs; ++epoch) {
172 // Forward pass
173 Matrix<T> Z = this->predict(X);
174
175 // Calculate loss
176 T loss = this->loss_function.forward(Z, y);
177
178 // Log progress if verbose
179 if (verbose) std::cout << "Epoch " << epoch + 1 << "/" << epochs << ", Loss: " << loss << std::endl;
180
181 // Calculate gradient of the loss
182 Matrix<T> gradient = this->loss_function.backward(Z, y);
183
184 // Backward propagation through all layers
185 for (int i = n_layers - 1; i >= 0; --i) {
186 gradient = backward_pass(gradient, layers[i]);
187 }
188
189 // Update weights and biases for all layers
190 for (int i = n_layers - 1; i >= 0; --i) {
191 optimizer.update_weights(layers[i].W, layers[i].dW, layers[i]._id);
192 optimizer.update_bias(layers[i].b, layers[i].db, layers[i]._id);
193
194 // Update normalization parameters if present
195 if (layers[i].normalization != nullptr) {
196 // Check if it's LayerNormalization (has gamma/beta)
197 auto* layer_norm = dynamic_cast<LayerNormalization<T>*>(layers[i].normalization);
198 if (layer_norm != nullptr) {
199 optimizer.update_weights(layer_norm->gamma, layer_norm->d_gamma, layers[i]._id + "_gamma");
200 optimizer.update_bias(layer_norm->beta, layer_norm->d_beta, layers[i]._id + "_beta");
201 }
202 }
203
204 optimizer.next_step();
205 }
206 }

288 static Matrix<T> forward_pass(const Matrix<T>& X, NeuralLayer<T>& layer) {
289 if (X.cols() != layer.W.rows()) {
290 throw std::invalid_argument("Incompatible dimensions for layer forward pass");
291 }
292 // Cache input for back-propagation
293 layer.X = X;
294 // Z = X * W + b
295 layer.Z = X * layer.W;
296 layer.Z.broadcast_horizontal_sum_inplace(layer.b);
297
298 // Apply normalization if present
299 Matrix<T> Z_normalized = layer.Z;
300 if (layer.normalization != nullptr) {
301 Z_normalized = layer.normalization->forward(layer.Z);
302 }
303
304 // A = activation(Z) - cache the activation output
305 layer.A = layer.activation.forward(Z_normalized);
306 return layer.A;
307 }

326 static Matrix<T> backward_pass(Matrix<T>& gradient, NeuralLayer<T>& layer) {
327 // dZ = dL/dZ (through activation)
328 // Note: If normalization was applied, we need to use the normalized Z
329 Matrix<T> Z_for_backward = layer.Z;
330 if (layer.normalization != nullptr) {
331 Z_for_backward = layer.normalization->forward(layer.Z); // Re-compute normalized Z
332 }
333
334 Matrix<T> dZ = layer.activation.backward(Z_for_backward, gradient);
335
336 // Backprop through normalization if present
337 if (layer.normalization != nullptr) {
338 dZ = layer.normalization->backward(dZ);
339 }
340
341 // prev_gradient = delta * W^T
342 Matrix<T> prev_gradient = dZ * layer.W.transpose();
343
344 // Get gradients for weights and bias updates
345 layer.db = dZ.horizontal_sum(); // Sum over rows to get bias gradient
346 layer.dW = layer.X.transpose() * dZ;
347
348 return prev_gradient;
349 }
